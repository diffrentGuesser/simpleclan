    @Override
    public void onInitialize() {
        System.out.println("[SimpleClan] Fabric mod initializing...");
        config = ClanConfig.load();
        CommandRegistrationCallback.EVENT.register((dispatcher, registryAccess, environment) -> this.registerCommands(dispatcher));
        ServerLifecycleEvents.SERVER_STOPPING.register(server -> {
            System.out.println("[SimpleClan] Server stopping. Saving clan data...");
            this.saveAllData();
        });
        ServerLifecycleEvents.SERVER_STARTED.register(server -> {
            dataFolder = Paths.get("clan_data");
            this.loadAllData();
            this.restoreAllTeams(server);
            this.rebuildPlayerClansMap();
        });
        ServerPlayConnectionEvents.JOIN.register((handler, sender, serverInstance) -> {
            ServerPlayer player = handler.getPlayer();
            this.applyTeamForPlayer(serverInstance, player, true);
            teamRecheck.put(player.getUUID(), 20);
        });
        ServerTickEvents.END_SERVER_TICK.register(this::tick);
    }

    /** Called by the locator bar mixin: may {@code receiver} see {@code source} on their locator bar? */
    public static boolean locatorVisible(ServerPlayer source, ServerPlayer receiver) {
        return switch (config.locatorBarMode()) {
            case EVERYONE -> true;
            case NOBODY -> false;
            case CLAN -> {
                String a = playerClans.get(source.getUUID());
                yield a != null && a.equals(playerClans.get(receiver.getUUID()));
            }
        };
    }

    /** Rebuilds every player's locator bar connections after clan membership or the mode changes. */
    private static void refreshLocatorBars(MinecraftServer server) {
        for (ServerLevel level : server.getAllLevels()) {
            for (ServerPlayer p : level.players()) {
                level.getWaypointManager().remakeConnections(p);
            }
        }
    }

    private void tick(MinecraftServer server) {
        if (!teamRecheck.isEmpty()) {
            for (UUID id : new ArrayList<>(teamRecheck.keySet())) {
                int left = teamRecheck.get(id) - 1;
                if (left > 0) {
                    teamRecheck.put(id, left);
                    continue;
                }
                teamRecheck.remove(id);
                ServerPlayer player = server.getPlayerList().getPlayer(id);
                if (player != null) {
                    this.applyTeamForPlayer(server, player, false);
                }
            }
        }
        // Save every 10 seconds (only files that changed), so clans survive a
        // server that is killed instead of stopped cleanly.
        if (dataFolder != null && --ticksUntilSave <= 0) {
            ticksUntilSave = 200;
            this.saveAllData(false);
        }
    }

    private void writeIfChanged(Path file, String json) throws java.io.IOException {
        String key = file.toString();
        if (json.equals(lastWritten.get(key)) && Files.exists(file)) {
            return;
        }
        Files.writeString(file, json);
        lastWritten.put(key, json);
    }

    private void saveAllData(boolean log) {
        try {
            if (!Files.exists(dataFolder)) {
                Files.createDirectories(dataFolder);
            }
​
            Path clansFile = dataFolder.resolve("clans.json");
            ClanDataContainer clanContainer = new ClanDataContainer();
            clanContainer.clans = new ArrayList<>(clans.values());
            this.writeIfChanged(clansFile, gson.toJson(clanContainer));
            Path invitesFile = dataFolder.resolve("invites.json");
            InviteDataContainer inviteContainer = new InviteDataContainer();
            inviteContainer.invites = new ArrayList<>();
​
            for (Entry<UUID, UUID> entry : pendingInvites.entrySet()) {
                inviteContainer.invites.add(new InviteEntry(entry.getKey(), entry.getValue(),
                        pendingInviteClans.get(entry.getKey())));
            }
​
            this.writeIfChanged(invitesFile, gson.toJson(inviteContainer));
            Path requestsFile = dataFolder.resolve("requests.json");
            RequestDataContainer requestContainer = new RequestDataContainer();
            requestContainer.requests = new ArrayList<>();
​
            for (Entry<String, List<UUID>> entry : joinRequests.entrySet()) {
                requestContainer.requests.add(new RequestEntry(entry.getKey(), entry.getValue()));
            }
​
            this.writeIfChanged(requestsFile, gson.toJson(requestContainer));
            if (log) {
                System.out.println("[SimpleClan] Data saved successfully.");
            }
        } catch (Exception e) {
            System.err.println("[SimpleClan] Failed to save data: " + e.getMessage());
            e.printStackTrace();
        }
    }

    private void loadAllData() {
        try {
            if (!Files.exists(dataFolder)) {
                System.out.println("[SimpleClan] No data folder found, starting fresh.");
                return;
            }
​
            Path clansFile = dataFolder.resolve("clans.json");
            if (Files.exists(clansFile)) {
                ClanDataContainer clanContainer = gson.fromJson(Files.readString(clansFile), ClanDataContainer.class);
                if (clanContainer != null && clanContainer.clans != null) {
                    clans.clear();
                    for (Clan clan : clanContainer.clans) {
                        clans.put(clan.name.toLowerCase(Locale.ROOT), clan);
                    }
                    System.out.println("[SimpleClan] Loaded " + clans.size() + " clans.");
                }
            }
​
            Path invitesFile = dataFolder.resolve("invites.json");
            if (Files.exists(invitesFile)) {
                InviteDataContainer inviteContainer = gson.fromJson(Files.readString(invitesFile), InviteDataContainer.class);
                if (inviteContainer != null && inviteContainer.invites != null) {
                    pendingInvites.clear();
                    pendingInviteClans.clear();
                    for (InviteEntry entry : inviteContainer.invites) {
                        pendingInvites.put(entry.invitee, entry.inviter);
                        if (entry.clan != null) {
                            pendingInviteClans.put(entry.invitee, entry.clan);
                        }
                    }
                }
            }
​
            Path requestsFile = dataFolder.resolve("requests.json");
            if (Files.exists(requestsFile)) {
                RequestDataContainer requestContainer = gson.fromJson(Files.readString(requestsFile), RequestDataContainer.class);
                if (requestContainer != null && requestContainer.requests != null) {
                    joinRequests.clear();
                    for (RequestEntry entry : requestContainer.requests) {
                        joinRequests.put(entry.clanName, entry.requests);
                    }
                }
            }
        } catch (Exception e) {
            System.err.println("[SimpleClan] Failed to load data: " + e.getMessage());
            e.printStackTrace();
        }
    }

    private void rebuildPlayerClansMap() {
        playerClans.clear();
        for (Entry<String, Clan> entry : clans.entrySet()) {
            for (UUID memberId : entry.getValue().members) {
                playerClans.put(memberId, entry.getKey());
            }
        }
        System.out.println("[SimpleClan] Rebuilt player-clan mapping for " + playerClans.size() + " players.");
    }

    private int cmdCreate(CommandContext<CommandSourceStack> ctx) {
        ServerPlayer player = this.getPlayer(ctx);
        if (player == null) {
            return 0;
        }
​
        UUID playerId = player.getUUID();
        String name = StringArgumentType.getString(ctx, "name");
        String colorArg = StringArgumentType.getString(ctx, "color").toUpperCase(Locale.ROOT);
        if (playerClans.containsKey(playerId)) {
            msg(player, red("You are already in a clan."));
            return 1;
        }
​
        if (name.length() > 16) {
            msg(player, red("Clan name too long (max 16 chars)."));
            return 1;
        }
​
        ChatFormatting color;
        try {
            color = ChatFormatting.valueOf(colorArg);
        } catch (Exception ex) {
            msg(player, red("Invalid color. Example: RED, BLUE, GREEN"));
            return 1;
        }
​
        String key = name.toLowerCase(Locale.ROOT);
        if (clans.containsKey(key)) {
            msg(player, red("A clan with that name already exists."));
            return 1;
        }
​
        Clan clan = new Clan(name, color, playerId);
        clan.members.add(playerId);
        clans.put(key, clan);
        playerClans.put(playerId, key);
        this.applyTeamForClan(player.level().getServer(), clan);
        refreshLocatorBars(player.level().getServer());
        msg(player, Component.literal("Clan created: ").append(Component.literal(name).withStyle(color)));
        return 1;
    }

    private int cmdInvite(CommandContext<CommandSourceStack> ctx) {
        ServerPlayer player = this.getPlayer(ctx);
        if (player == null) {
            return 0;
        }
        UUID playerId = player.getUUID();
        String targetName = StringArgumentType.getString(ctx, "player");
        String clanKey = playerClans.get(playerId);
        if (clanKey == null) {
            msg(player, red("You are not in a clan."));
            return 1;
        }
        Clan clan = clans.get(clanKey);
        if (!clan.owner.equals(playerId)) {
            msg(player, red("Only the owner can invite players."));
            return 1;
        }
        ServerPlayer target = this.lookupOnlinePlayer(player.level().getServer(), targetName);
        if (target == null) {
            msg(player, red("Player not found."));
            return 1;
        }
        if (playerClans.containsKey(target.getUUID())) {
            msg(player, red("That player is already in a clan."));
            return 1;
        }
        pendingInvites.put(target.getUUID(), playerId);
        pendingInviteClans.put(target.getUUID(), clanKey);
        msg(target, Component.literal(player.getName().getString() + " invited you to join clan ")
                .append(Component.literal(clan.name).withStyle(clan.color))
                .append(Component.literal(". Type /clan accept to join or /clan deny to decline."))
                .withStyle(ChatFormatting.YELLOW));
        msg(player, green("Invite sent to " + target.getName().getString()));
        return 1;
    }

    private int cmdAccept(CommandContext<CommandSourceStack> ctx) {
        ServerPlayer player = this.getPlayer(ctx);
        if (player == null) {
            return 0;
        }
​
        UUID playerId = player.getUUID();
        if (!pendingInvites.containsKey(playerId)) {
            msg(player, red("You have no pending invites."));
            return 1;
        }
​
        UUID inviterId = pendingInvites.remove(playerId);
        String inviteClan = pendingInviteClans.remove(playerId);
        // Invites saved by older versions have no clan key; fall back to the inviter's clan.
        String clanKey = inviteClan != null ? inviteClan : playerClans.get(inviterId);
        if (clanKey == null || !clans.containsKey(clanKey)) {
            msg(player, red("The inviting clan no longer exists."));
            return 1;
        }
        if (playerClans.containsKey(playerId)) {
            msg(player, red("You are already in a clan."));
            return 1;
        }
​
        Clan clan = clans.get(clanKey);
        clan.members.add(playerId);
        playerClans.put(playerId, clanKey);
        MinecraftServer server = player.level().getServer();
        this.applyTeamForClan(server, clan);
        refreshLocatorBars(server);
        msg(player, Component.literal("You joined clan ").append(Component.literal(clan.name).withStyle(clan.color)));
        ServerPlayer inviter = server.getPlayerList().getPlayer(inviterId);
        if (inviter != null) {
            msg(inviter, green(player.getName().getString() + " joined your clan."));
        }
        return 1;
    }

    private int cmdAcceptPlayer(CommandContext<CommandSourceStack> ctx) {
        ServerPlayer player = this.getPlayer(ctx);
        if (player == null) {
            return 0;
        }
​
        UUID playerId = player.getUUID();

            return 1;

​
        requests.remove(found);
        joinRequests.put(ownerClanKey, requests);
        if (playerClans.containsKey(found)) {
            msg(player, red("That player is already in a clan."));
            return 1;
        }
​
        clan.members.add(found);
        playerClans.put(found, ownerClanKey);
        this.applyTeamForClan(server, clan);
        refreshLocatorBars(server);
        ServerPlayer newly = server.getPlayerList().getPlayer(found);
        if (newly != null) {
            msg(newly, Component.literal("Your request to join clan ")
                    .append(Component.literal(clan.name).withStyle(clan.color))
                    .append(Component.literal(" was accepted."))
                    .withStyle(ChatFormatting.GREEN));
        }
​
        msg(player, green("Player accepted into the clan."));

    private static UUID findRequest(MinecraftServer server, List<UUID> requests, String targetName) {
        for (UUID u : new ArrayList<>(requests)) {
            ServerPlayer possible = server.getPlayerList().getPlayer(u);
            if (possible != null && possible.getName().getString().equalsIgnoreCase(targetName)) {
                return u;
            }
        }
        return null;
    }

    private int cmdKick(CommandContext<CommandSourceStack> ctx) {
        ServerPlayer player = this.getPlayer(ctx);
        if (player == null) {
            return 0;
        }
        UUID playerId = player.getUUID();
        String targetName = StringArgumentType.getString(ctx, "player");
        String clanKey = playerClans.get(playerId);
        if (clanKey == null) {
            msg(player, red("You are not in a clan."));
            return 1;
        }
        Clan clan = clans.get(clanKey);
        if (!clan.owner.equals(playerId)) {
            msg(player, red("Only the owner can kick members."));
            return 1;
        }
        MinecraftServer server = player.level().getServer();
        ServerPlayer target = this.lookupOnlinePlayer(server, targetName);
        if (target == null) {
            msg(player, red("Player not found."));
            return 1;
        }
        if (!clan.members.contains(target.getUUID())) {
            msg(player, red("That player is not in your clan."));
            return 1;
        }
        if (target.getUUID().equals(clan.owner)) {
            msg(player, red("You can't kick yourself. Use /clan disband or /clan transfer."));
            return 1;
        }
        clan.members.remove(target.getUUID());
        playerClans.remove(target.getUUID());
        this.removeFromTeam(server, target);
        refreshLocatorBars(server);
        msg(player, green(target.getName().getString() + " kicked from clan."));
        msg(target, Component.literal("You were kicked from clan ")
                .append(Component.literal(clan.name).withStyle(clan.color))
                .withStyle(ChatFormatting.RED));
        return 1;
    }

    private void applyTeamForClan(MinecraftServer server, Clan clan) {
        Scoreboard sb = server.getScoreboard();
        String teamId = this.teamIdFromClanName(clan.name);
        PlayerTeam team = sb.getPlayerTeam(teamId);
        if (team == null) {
            team = sb.addPlayerTeam(teamId);
        }
​
        if (config.tagOnRight()) {
            team.setPlayerPrefix(Component.empty());
            team.setPlayerSuffix(Component.literal(" [" + clan.name + "]").withStyle(clan.color));
        } else {
            team.setPlayerPrefix(Component.literal("[" + clan.name + "] ").withStyle(clan.color));
            team.setPlayerSuffix(Component.empty());
        }
​
        // Only remove online players who are no longer members. Offline members
        // stay in the team, so their tag is already there when they rejoin.
        for (ServerPlayer online : server.getPlayerList().getPlayers()) {
            String name = online.getScoreboardName();
            if (team.getPlayers().contains(name) && !clan.members.contains(online.getUUID())) {
                sb.removePlayerFromTeam(name, team);
            }
        }
​
        for (UUID memberId : clan.members) {
            ServerPlayer player = server.getPlayerList().getPlayer(memberId);
            if (player != null && team != sb.getPlayersTeam(player.getScoreboardName())) {
                sb.addPlayerToTeam(player.getScoreboardName(), team);
            }
        }
    }

    /** Puts a player into their clan's team (or out of all clan teams), without needing the player list. */
    private void applyTeamForPlayer(MinecraftServer server, ServerPlayer player, boolean log) {
        String clanKey = playerClans.get(player.getUUID());
        Clan clan = clanKey != null ? clans.get(clanKey) : null;
        String name = player.getScoreboardName();
        Scoreboard sb = server.getScoreboard();
        if (clan == null) {
            PlayerTeam current = sb.getPlayersTeam(name);
            if (current != null && current.getName().startsWith("clan_")) {
                sb.removePlayerFromTeam(name, current);
            }
            return;
        }
        this.applyTeamForClan(server, clan);
        PlayerTeam team = sb.getPlayerTeam(this.teamIdFromClanName(clan.name));
        if (team != null && sb.getPlayersTeam(name) != team) {
            sb.addPlayerToTeam(name, team);
        }
        if (log) {
            System.out.println("[SimpleClan] Applied team for " + name + " in clan " + clan.name);
        }
    }
